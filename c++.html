    #include <iostream>
        
    int main (){
    
        char op;
        double num1;
        double num2;
        double result;

        std::cout << " Irjabe barmelyiket (+ - * /): ";
        std::cin >> op;
    
        std::cout << "Elso szam: ";
        std::cin >> num1;
    
        std::cout << "Masodik szam: ";
        std::cin >> num2;
    
        switch (op)
        {
        case '+':
            result = num1 + num2;
            std::cout <<"Eredmeny: " << result << '\n';
            break;
        case '-':
            result = num1 - num2;
            std::cout <<"Eredmeny: " << result << '\n';
            break;
        case '*':
            result = num1 * num2;
            std::cout <<"Eredmeny: " << result << '\n';
            break;
        case '/':
            result = num1 / num2;
            std::cout <<"Eredmeny: " << result << '\n';
            break;
        
        default:
            std::cout << "Ervenytelen adat";
            break;
        }
    
        std::cout << "******************************";
    
    
    
    
        return 0;
    }
